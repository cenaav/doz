# Doz Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the Doz app shell (Nuxt SSG, bilingual, themed, responsive, offline), the tested dose engine, the validated JSON data model, search, and 8 seed drugs that exercise every regimen type, deployed to GitHub Pages.

**Architecture:** Pure TypeScript engine/search/validation in `lib/` (no Vue), tested with Vitest. Data is JSON in `data/`, validated by Ajv + integrity checks in CI. Nuxt 4 app in `app/` pre-renders every page; drug files are lazy-loaded per page; a generated catalog powers search and lists.

**Tech Stack:** Nuxt 4.6, Vue 3.5, @nuxtjs/i18n 10, @nuxtjs/color-mode, Tailwind CSS 4.3 (@tailwindcss/vite), @vite-pwa/nuxt, Fuse.js 7, Ajv 8, Vitest, tsx, @fontsource-variable/vazirmatn.

**Spec:** `docs/superpowers/specs/2026-10-15-doz-design.md`

## Global Constraints

- Node 24; `npm` as package manager.
- Deployed at `https://cenaav.github.io/doz/`; build with `NUXT_APP_BASE_URL=/doz/`.
- Persian default locale (RTL, no prefix), English at `/en` (LTR).
- Font: Vazirmatn, self-hosted (no CDN).
- No patient data leaves the device; only `localStorage` for disclaimer, recent drugs, theme, language.
- Every regimen, limit, contraindication, warning, renal rule, monitoring item, interaction and note in data must have ≥ 1 ref; each ref has `section` or `page`.
- Nothing cited from memory: every FDA ref is checked against the live DailyMed label; every guideline page against the published article.
- Engine never returns a dose above an applicable cap without reporting the cap.
- Age is always required in the calculator.

## Review Focus

1. Persian digits / Persian decimal separator typed into age or weight → parsed correctly, not ignored (test in Task 7 `parseNumber`).
2. Age exactly on a band or population boundary (e.g. 2.0 y, 6 months) → falls in exactly one band, never none (test in Task 3 `inRange`/Task 4 ageBand).
3. Adult-sized weight on a pediatric mg/kg regimen → capped and cap shown (test in Task 3).
4. Combination product where the dosed ingredient is fine but the other ingredient exceeds its limit → blocked (test in Task 4).
5. Deep link opened directly (e.g. `/doz/en/drug/amoxicillin/`) on GitHub Pages → page renders, not 404 (check in Task 9 generate output).

- [ ] **Step 1: `lib/types.ts`**

```ts
export type Lang = 'fa' | 'en'
export type I18nText = Partial<Record<Lang, string>>
export type AgeUnit = 'day' | 'week' | 'month' | 'year'
export interface Age { value: number; unit: AgeUnit }

export interface Ref {
  source: string
  section?: string | null
  page?: string | null
  url?: string
  note?: I18nText
}

export type SourceType = 'fda-label' | 'ema-smpc' | 'guideline' | 'licensed-reference' | 'ifda-leaflet' | 'textbook'
export interface Source {
  id: string
  type: SourceType
  short: string
  title: string
  publisher?: string
  edition?: string
  year?: number
  citation?: string
  url?: string
  accessed?: string
  license: 'public' | 'licensed' | 'pending'
}

export interface Range { minAge?: Age | null; maxAge?: Age | null; minWeightKg?: number | null; maxWeightKg?: number | null }

export type DoseUnit = 'mg' | 'mcg' | 'g' | 'unit' | 'mL'
export interface Concentration { amount: number; unit: DoseUnit; perVolumeMl?: number; perUnit?: 'tablet' | 'capsule' | 'sachet'; scored?: boolean }
export interface Limits { maxPerDose?: number; maxPerDay?: number; maxPerKgPerDose?: number; maxPerKgPerDay?: number; refs: Ref[] }
export interface Component { name: I18nText; amount: number; unit: DoseUnit; limits?: Limits }
export interface Formulation {
  id: string; form: string; route: string; label: string
  concentration: Concentration
  components?: Record<string, Component> | null
  markets: string[]
  refs: Ref[]
}
export interface Note { text: I18nText; refs: Ref[] }

export interface AgeBandRow { minAge: Age; maxAge: Age | null; perDose: number; dosesPerDay: number; refs?: Ref[] }
export interface LoadingDay { fromDay: number; toDay: number; perKgPerDay: number; maxPerDay?: number }
export interface TitrationSteps { start: number; increment?: number; incrementEveryDays?: number; targetMin?: number; targetMax?: number; max: number }
export type RegimenType = 'weightBased' | 'fixed' | 'ageBand' | 'bsa' | 'loading' | 'titration' | 'notCalculated'
export interface Regimen {
  id: string
  disease: string
  label: I18nText
  population: Range
  type: RegimenType
  unit?: DoseUnit
  dose: {
    perKgPerDay?: number; perKgPerDose?: number; dividedDoses?: number; intervalHours?: number
    perDose?: number; dosesPerDay?: number
    perM2PerDay?: number
    bands?: AgeBandRow[]
    days?: LoadingDay[]
    titration?: TitrationSteps
    weightBands?: (TitrationSteps & { minWeightKg: number; maxWeightKg: number | null })[]
    reason?: I18nText
  }
  doseComponent?: string
  limits?: Limits
  durationDays?: { min: number; max: number } | null
  allowedFormulations?: string[]
  refs: Ref[]
  notes?: Note[]
}

export interface Contraindication { id: string; severity: 'block' | 'warn'; rule: Range; text: I18nText; refs: Ref[] }
export interface RenalRule { egfrBelow: number; severity: 'block' | 'warn'; text: I18nText; refs: Ref[] }
export interface Example {
  regimen: string; formulation: string
  input: { ageDays?: number; ageMonths?: number; ageYears?: number; weightKg: number; heightCm?: number; scrMgDl?: number }
  expect: Partial<Record<'status' | 'perDose' | 'perDay' | 'dosesPerDay' | 'mlPerDose' | 'mlPerDoseRounded' | 'tabletsPerDoseRounded', number | string>>
}
export interface Drug {
  id: string; schemaVersion: 1
  name: I18nText; category: string; class?: I18nText
  brands: { name: string; market: string }[]
  synonyms: string[]
  formulations: Formulation[]
  regimens: Regimen[]
  contraindications: Contraindication[]
  warnings: Note[]
  renal: RenalRule[]
  monitoring: Note[]
  interactions: (Note & { with: string })[]
  review: { status: 'unverified' | 'verified'; reviewer: string | null; date: string | null }
  examples: Example[]
}
export interface Disease { id: string; name: I18nText; synonyms: string[]; icd10: string[]; specialty: string }
export interface Category { id: string; name: I18nText; icon: string; order: number }
export interface CatalogEntry {
  id: string; category: string; name: I18nText; class?: I18nText
  brands: string[]; synonyms: string[]
  reviewStatus: 'unverified' | 'verified'
  regimens: { id: string; disease: string; label: I18nText }[]
}
export interface DataSet { sources: Source[]; diseases: Disease[]; categories: Category[]; drugs: { path: string; drug: Drug }[] }
```

```ts
import type { Drug, Ref, Source, SourceType } from '../types'

export function collectRefs(node: unknown, out: Ref[] = []): Ref[] {
  if (Array.isArray(node)) node.forEach((n) => collectRefs(n, out))
  else if (node && typeof node === 'object') {
    for (const [k, v] of Object.entries(node)) {
      if (k === 'refs' && Array.isArray(v)) out.push(...(v as Ref[]))
      else collectRefs(v, out)
    }
  }
  return out
}

export function refKey(r: Ref): string {
  return `${r.source}|${r.section ?? ''}|${r.page ?? ''}`
}

export function uniqueRefs(refs: Ref[]): Ref[] {
  const seen = new Set<string>()
  return refs.filter((r) => (seen.has(refKey(r)) ? false : (seen.add(refKey(r)), true)))
}

export function groupRefsByType(refs: Ref[], sources: Source[]): Map<SourceType, Ref[]> {
  const byId = new Map(sources.map((s) => [s.id, s]))
  const map = new Map<SourceType, Ref[]>()
  for (const r of uniqueRefs(refs)) {
    const s = byId.get(r.source)
    if (!s) continue
    map.set(s.type, [...(map.get(s.type) ?? []), r])
  }
  return map
}

export function drugRefs(drug: Drug): Ref[] {
  return uniqueRefs(collectRefs(drug))
}
```

```ts
function walk(dir: string): string[] {
  return readdirSync(dir).flatMap((f) => {
    const p = join(dir, f)
    return statSync(p).isDirectory() ? walk(p) : p.endsWith('.json') ? [p] : []
  })

  const dataDir = join(root, 'data')
  return {
    sources: readJson(join(dataDir, 'sources.json')),
    diseases: readJson(join(dataDir, 'diseases.json')),
    categories: readJson(join(dataDir, 'categories.json')),
    drugs: walk(join(dataDir, 'drugs')).sort().map((p) => ({ path: relative(dataDir, p), drug: readJson(p) as Drug })),
```

```ts
import Ajv2020 from 'ajv/dist/2020'
import addFormats from 'ajv-formats'
import { readFileSync } from 'node:fs'
import { join } from 'node:path'
import type { DataSet, Drug } from '../types'
import { collectRefs } from './refs'

export function validateDataSet(ds: DataSet, root: string): string[] {
  const errors: string[] = []
  const ajv = new Ajv2020({ allErrors: true })
  addFormats(ajv)
  const schema = (n: string) => ajv.compile(JSON.parse(readFileSync(join(root, 'schema', `${n}.schema.json`), 'utf8')))
  const check = (name: string, fn: ReturnType<typeof schema>, value: unknown) => {
    if (!fn(value)) for (const e of fn.errors ?? []) errors.push(`${name}: ${e.instancePath || '/'} ${e.message}`)
  }
  check('sources.json', schema('source'), ds.sources)
  check('diseases.json', schema('disease'), ds.diseases)
  check('categories.json', schema('category'), ds.categories)
  const drugSchema = schema('drug')
  for (const { path, drug } of ds.drugs) check(path, drugSchema, drug)
  if (errors.length) return errors

  const sourceIds = new Set(ds.sources.map((s) => s.id))
  const diseaseIds = new Set(ds.diseases.map((d) => d.id))
  const categoryIds = new Set(ds.categories.map((c) => c.id))
  const dup = (ids: string[], what: string) => {
    const seen = new Set<string>()
    for (const id of ids) { if (seen.has(id)) errors.push(`duplicate ${what} id: ${id}`); seen.add(id) }
  }
  dup(ds.sources.map((s) => s.id), 'source')
  dup(ds.diseases.map((d) => d.id), 'disease')
  dup(ds.categories.map((c) => c.id), 'category')
  dup(ds.drugs.map((d) => d.drug.id), 'drug')

  for (const { path, drug } of ds.drugs) errors.push(...checkDrug(path, drug, sourceIds, diseaseIds, categoryIds))
  return errors
}

function checkDrug(path: string, d: Drug, sourceIds: Set<string>, diseaseIds: Set<string>, categoryIds: Set<string>): string[] {
  const e: string[] = []
  const expectedPath = `drugs/${d.category}/${d.id}.json`
  if (path.replaceAll('\\', '/') !== expectedPath) e.push(`${path}: must be at ${expectedPath}`)
  if (!categoryIds.has(d.category)) e.push(`${path}: unknown category ${d.category}`)
  for (const r of collectRefs(d)) if (!sourceIds.has(r.source)) e.push(`${path}: unknown source ${r.source}`)

  const formIds = new Set(d.formulations.map((f) => f.id))
  const regIds = new Set(d.regimens.map((r) => r.id))
  for (const r of d.regimens) {
    if (!diseaseIds.has(r.disease)) e.push(`${path}: regimen ${r.id} unknown disease ${r.disease}`)
    for (const f of r.allowedFormulations ?? []) if (!formIds.has(f)) e.push(`${path}: regimen ${r.id} unknown formulation ${f}`)
    const unit = r.unit ?? 'mg'
    if (r.type !== 'notCalculated' && unit !== 'mL') {
      for (const f of d.formulations.filter((f) => !r.allowedFormulations || r.allowedFormulations.includes(f.id)))
        if (f.concentration.unit !== unit) e.push(`${path}: regimen ${r.id} unit ${unit} ≠ formulation ${f.id} unit ${f.concentration.unit}`)
    }
    if (r.doseComponent) {
      for (const f of d.formulations) if (f.components && !f.components[r.doseComponent]) e.push(`${path}: formulation ${f.id} lacks component ${r.doseComponent}`)
    }
    if (r.type !== 'notCalculated' && !d.examples.some((x) => x.regimen === r.id)) e.push(`${path}: regimen ${r.id} has no example`)
  }
  for (const x of d.examples) {
    if (!regIds.has(x.regimen)) e.push(`${path}: example unknown regimen ${x.regimen}`)
    if (!formIds.has(x.formulation)) e.push(`${path}: example unknown formulation ${x.formulation}`)
    if (x.input.ageDays == null && x.input.ageMonths == null && x.input.ageYears == null) e.push(`${path}: example needs an age`)
  }
  if (d.review.status === 'verified' && (!d.review.reviewer || !d.review.date)) e.push(`${path}: verified needs reviewer and date`)
  return e
}
```

```ts
import type { Age, AgeUnit, Range } from '../types'
import type { Patient } from './index'

const DAYS: Record<AgeUnit, number> = { day: 1, week: 7, month: 30.4375, year: 365.25 }
export const toDays = (a: Age) => a.value * DAYS[a.unit]

export function ageDaysFrom(i: { ageDays?: number; ageMonths?: number; ageYears?: number }): number {
  if (i.ageDays != null) return i.ageDays
  if (i.ageMonths != null) return i.ageMonths * DAYS.month
  return (i.ageYears ?? NaN) * DAYS.year
}

export function inRange(p: Patient, r: Range): boolean {
  if (r.minAge && p.ageDays < toDays(r.minAge)) return false
  if (r.maxAge && p.ageDays >= toDays(r.maxAge)) return false
  if (r.minWeightKg != null && p.weightKg < r.minWeightKg) return false
  if (r.maxWeightKg != null && p.weightKg >= r.maxWeightKg) return false
  return true
}
```

```ts
export const r2 = (x: number) => Math.round(x * 100) / 100
const snap = (x: number, step: number) => r2(Math.round(Number((x / step).toFixed(6))) * step)
export function roundMl(ml: number): number {
  return snap(ml, ml < 1 ? 0.05 : ml <= 10 ? 0.1 : 0.5)
}
export function roundTablets(n: number, scored: boolean): number {
  return scored ? snap(n, 0.5) : Math.round(n)
```

```ts
import type { DoseUnit, Limits } from '../types'
import type { CapHit } from './index'

export function applyLimits(perDose: number, dosesPerDay: number, weightKg: number, limits: Limits | undefined, unit: DoseUnit) {
  const hits: CapHit[] = []
  if (!limits) return { perDose, perDay: perDose * dosesPerDay, hits }
  const hit = (kind: CapHit['kind'], limit: number) => hits.push({ kind, limit, unit, refs: limits.refs })
  const doseCaps: [CapHit['kind'], number | undefined][] = [
    ['maxPerKgPerDose', limits.maxPerKgPerDose != null ? limits.maxPerKgPerDose * weightKg : undefined],
    ['maxPerDose', limits.maxPerDose],
  ]
  for (const [kind, cap] of doseCaps) if (cap != null && perDose > cap) { perDose = cap; hit(kind, cap) }
  let perDay = perDose * dosesPerDay
  const dayCaps: [CapHit['kind'], number | undefined][] = [
    ['maxPerKgPerDay', limits.maxPerKgPerDay != null ? limits.maxPerKgPerDay * weightKg : undefined],
    ['maxPerDay', limits.maxPerDay],
  ]
  for (const [kind, cap] of dayCaps) if (cap != null && perDay > cap) { perDay = cap; perDose = cap / dosesPerDay; hit(kind, cap) }
  return { perDose, perDay, hits }
}
```

```ts
import type { Msg, Patient } from './index'

// [ageYears, minKg, maxKg]; linear interpolation between rows. Heuristic for typo detection only.
const BOUNDS: [number, number, number][] = [
  [0, 1.7, 5.3], [0.5, 4.7, 11.5], [1, 6, 14.4], [2, 7.7, 17.8], [3, 8.9, 21.3], [5, 11.5, 28.2],
  [8, 15.7, 42.6], [10, 18.7, 55.2], [12, 23, 71.3], [15, 31.5, 92], [18, 35.7, 109],
]

export function plausibilityWarnings(p: Patient): Msg[] {
  const y = p.ageDays / 365.25
  if (y >= 18) return []
  const i = BOUNDS.findIndex(([a]) => a > y)
  const [a0, lo0, hi0] = BOUNDS[i - 1]
  const [a1, lo1, hi1] = BOUNDS[i]
  const t = (y - a0) / (a1 - a0)
  const lo = lo0 + t * (lo1 - lo0)
  const hi = hi0 + t * (hi1 - hi0)
  return p.weightKg < lo || p.weightKg > hi
    ? [{ code: 'WEIGHT_IMPLAUSIBLE', params: { min: Math.round(lo * 10) / 10, max: Math.round(hi * 10) / 10 } }]
    : []
}
```

```ts
import type { Drug, Regimen } from '../types'
import { toDays } from './age'
import type { Msg, Patient, ScheduleStep, TitrationPlan } from './index'

export interface BaseDose {
  perDose?: number; dosesPerDay?: number
  schedule?: Omit<ScheduleStep, 'mlPerDose' | 'mlPerDoseRounded'>[]
  titration?: TitrationPlan
  bsa?: number
  error?: Msg
  extraRefs?: Drug['regimens'][number]['refs']
}

export function baseDose(reg: Regimen, p: Patient): BaseDose {
  const d = reg.dose
  switch (reg.type) {
    case 'weightBased':
      if (d.perKgPerDay != null) return { perDose: (d.perKgPerDay * p.weightKg) / d.dividedDoses!, dosesPerDay: d.dividedDoses }
      return { perDose: d.perKgPerDose! * p.weightKg, dosesPerDay: 24 / d.intervalHours! }
    case 'fixed':
      return { perDose: d.perDose, dosesPerDay: d.dosesPerDay }
    case 'bsa': {
      if (!p.heightCm) return { error: { code: 'NEED_HEIGHT' } }
      const bsa = Math.sqrt((p.heightCm * p.weightKg) / 3600)
      return { perDose: (d.perM2PerDay! * bsa) / d.dividedDoses!, dosesPerDay: d.dividedDoses, bsa }
    }
    case 'ageBand': {
      const b = d.bands!.find((x) => p.ageDays >= toDays(x.minAge) && (!x.maxAge || p.ageDays < toDays(x.maxAge)))
      if (!b) return { error: { code: 'NO_AGE_BAND' } }
      return { perDose: b.perDose, dosesPerDay: b.dosesPerDay, extraRefs: b.refs }
    }
    case 'loading': {
      const n = d.dividedDoses!
      const schedule = d.days!.map((x) => {
        const raw = x.perKgPerDay * p.weightKg
        const perDay = x.maxPerDay != null ? Math.min(raw, x.maxPerDay) : raw
        return { fromDay: x.fromDay, toDay: x.toDay, perDay, perDose: perDay / n, capped: perDay < raw }
      })
      return { perDose: schedule[0].perDose, dosesPerDay: n, schedule }
    }
    case 'titration': {
      const steps = d.titration ?? d.weightBands!.find((w) => p.weightKg >= w.minWeightKg && (w.maxWeightKg == null || p.weightKg < w.maxWeightKg))
      if (!steps) return { error: { code: 'NO_WEIGHT_BAND' } }
      const n = d.dividedDoses!
      const { start, increment, incrementEveryDays, targetMin, targetMax, max } = steps
      return { perDose: start / n, dosesPerDay: n, titration: { start, increment, incrementEveryDays, targetMin, targetMax, max, dividedDoses: n } }
    }
    default:
      return {}
  }
}
```

```ts
import type { DoseUnit, Drug, I18nText, Ref } from '../types'
import { uniqueRefs } from '../data/refs'
import { inRange } from './age'
import { applyLimits } from './caps'
import { plausibilityWarnings } from './plausibility'
import { baseDose } from './regimens'
import { schwartzEgfr } from './renal'
import { r2, roundMl, roundTablets } from './rounding'

export interface Patient { ageDays: number; weightKg: number; heightCm?: number; scrMgDl?: number }
export interface Msg { code: string; text?: I18nText; refs?: Ref[]; params?: Record<string, string | number> }
export interface CapHit { kind: 'maxPerKgPerDose' | 'maxPerDose' | 'maxPerKgPerDay' | 'maxPerDay'; limit: number; unit: DoseUnit; refs: Ref[] }
export interface ScheduleStep { fromDay: number; toDay: number; perDay: number; perDose: number; mlPerDose?: number; mlPerDoseRounded?: number; capped: boolean }
export interface TitrationPlan { start: number; increment?: number; incrementEveryDays?: number; targetMin?: number; targetMax?: number; max: number; dividedDoses: number }
export interface Result {
  status: 'ok' | 'blocked' | 'notCalculated' | 'invalidInput'
  unit: DoseUnit
  perDose?: number; perDay?: number; dosesPerDay?: number; intervalHours?: number
  mlPerDose?: number; mlPerDoseRounded?: number; mlPerDay?: number
  tabletsPerDose?: number; tabletsPerDoseRounded?: number; tabletUnit?: string
  durationDays?: { min: number; max: number } | null
  schedule?: ScheduleStep[]; titration?: TitrationPlan
  capsApplied: CapHit[]; warnings: Msg[]; blocks: Msg[]; refs: Ref[]
  egfr?: number; bsa?: number; reason?: I18nText
}

const MAX_AGE_DAYS = 120 * 365.25

export function calculate(drug: Drug, regimenId: string, formulationId: string, p: Patient): Result {
  const reg = drug.regimens.find((r) => r.id === regimenId)
  const form = drug.formulations.find((f) => f.id === formulationId)
  const unit = reg?.unit ?? 'mg'
  const res: Result = { status: 'ok', unit, capsApplied: [], warnings: [], blocks: [], refs: [] }
  const invalid = (code: string): Result => ({ ...res, status: 'invalidInput', blocks: [{ code }] })

  if (!reg) return invalid('UNKNOWN_REGIMEN')
  if (!form) return invalid('UNKNOWN_FORMULATION')
  if (!Number.isFinite(p.weightKg) || p.weightKg <= 0 || p.weightKg > 300) return invalid('INVALID_WEIGHT')
  if (!Number.isFinite(p.ageDays) || p.ageDays < 0 || p.ageDays > MAX_AGE_DAYS) return invalid('INVALID_AGE')

  const refs: Ref[] = [...reg.refs, ...form.refs]
  res.warnings.push(...plausibilityWarnings(p))

  for (const c of drug.contraindications) {
    if (!inRange(p, c.rule)) continue
    const msg = { code: 'CONTRAINDICATION', text: c.text, refs: c.refs }
    ;(c.severity === 'block' ? res.blocks : res.warnings).push(msg)
    refs.push(...c.refs)
  }
  if (p.heightCm && p.scrMgDl) {
    if (p.ageDays < 18 * 365.25) {
      res.egfr = r2(schwartzEgfr(p.heightCm, p.scrMgDl))
      for (const rr of drug.renal) {
        if (res.egfr >= rr.egfrBelow) continue
        ;(rr.severity === 'block' ? res.blocks : res.warnings).push({ code: 'RENAL', text: rr.text, refs: rr.refs, params: { egfr: res.egfr } })
        refs.push(...rr.refs)
      }
    } else res.warnings.push({ code: 'EGFR_ADULT_NOT_COMPUTED' })
  }
  if (reg.allowedFormulations && !reg.allowedFormulations.includes(form.id)) res.blocks.push({ code: 'FORMULATION_NOT_ALLOWED', refs: reg.refs })
  if (!inRange(p, reg.population)) res.blocks.push({ code: 'OUTSIDE_POPULATION', refs: reg.refs })

  const done = (status: Result['status']): Result => ({ ...res, status, refs: uniqueRefs(refs) })
  if (res.blocks.length) return done('blocked')
  if (reg.type === 'notCalculated') return { ...done('notCalculated'), reason: reg.dose.reason }

  const base = baseDose(reg, p)
  if (base.error) {
    if (base.error.code === 'NEED_HEIGHT') return invalid('NEED_HEIGHT')
    res.blocks.push({ ...base.error, refs: reg.refs })
    return done('blocked')
  }
  if (base.extraRefs) refs.push(...base.extraRefs)
  if (reg.limits) refs.push(...reg.limits.refs)

  const n = base.dosesPerDay!
  const capped = applyLimits(base.perDose!, n, p.weightKg, reg.limits, unit)
  res.capsApplied = capped.hits
  res.perDose = r2(capped.perDose)
  res.perDay = r2(capped.perDay)
  res.dosesPerDay = r2(n)
  res.intervalHours = r2(24 / n)
  res.durationDays = reg.durationDays ?? null
  res.bsa = base.bsa
  res.titration = base.titration

  // other-ingredient limits for combination products
  if (form.components && reg.doseComponent) {
    for (const [key, comp] of Object.entries(form.components)) {
      if (key === reg.doseComponent || !comp.limits) continue
      const compPerDay = (capped.perDay * comp.amount) / form.concentration.amount
      const l = comp.limits
      const over = (l.maxPerKgPerDay != null && compPerDay > l.maxPerKgPerDay * p.weightKg) || (l.maxPerDay != null && compPerDay > l.maxPerDay)
      if (over) {
        res.blocks.push({ code: 'COMPONENT_LIMIT', text: comp.name, refs: l.refs, params: { perDay: r2(compPerDay), unit: comp.unit } })
        refs.push(...l.refs)
      }
    }
    if (res.blocks.length) {
      const { perDose, perDay, dosesPerDay, intervalHours, ...rest } = res
      return { ...rest, status: 'blocked', refs: uniqueRefs(refs) }
    }
  }

  const c = form.concentration
  const toMl = (amount: number) => (unit === 'mL' ? amount : (amount * c.perVolumeMl!) / c.amount)
  if (c.perVolumeMl != null || unit === 'mL') {
    res.mlPerDose = r2(toMl(capped.perDose))
    res.mlPerDoseRounded = roundMl(toMl(capped.perDose))
    res.mlPerDay = r2(toMl(capped.perDay))
    res.schedule = base.schedule?.map((s) => ({ ...s, perDay: r2(s.perDay), perDose: r2(s.perDose), mlPerDose: r2(toMl(s.perDose)), mlPerDoseRounded: roundMl(toMl(s.perDose)) }))
  } else {
    const t = capped.perDose / c.amount
    res.tabletsPerDose = r2(t)
    res.tabletsPerDoseRounded = roundTablets(t, !!c.scored)
    res.tabletUnit = c.perUnit
    if (res.tabletsPerDoseRounded === 0 || Math.abs(res.tabletsPerDoseRounded - t) / t > 0.1)
      res.warnings.push({ code: 'TABLET_ROUNDING', params: { exact: r2(t), rounded: res.tabletsPerDoseRounded } })
    res.schedule = base.schedule?.map((s) => ({ ...s, perDay: r2(s.perDay), perDose: r2(s.perDose) }))
  }
  return done('ok')
}
```

```ts
const FA = '۰۱۲۳۴۵۶۷۸۹'
const AR = '٠١٢٣٤٥٦٧٨٩'
export function normalize(s: string): string {
  return s
    .replace(/[يى]/g, 'ی')
    .replace(/ك/g, 'ک')
    .replace(/ة/g, 'ه')
    .replace(/[أإآ]/g, 'ا')
    .replace(/[ً-ٰٟـ]/g, '')
    .replace(/[‌‍‎‏]/g, ' ')
    .replace(/[۰-۹]/g, (d) => String(FA.indexOf(d)))
    .replace(/[٠-٩]/g, (d) => String(AR.indexOf(d)))
    .toLowerCase()
    .replace(/\s+/g, ' ')
    .trim()
}
```

```ts
import Fuse from 'fuse.js'
import type { CatalogEntry, Category, Disease } from '../types'
import { normalize } from './normalize'

export interface SearchResults { drugs: CatalogEntry[]; diseases: Disease[]; categories: Category[] }
type Doc = { kind: 'drug' | 'disease' | 'category'; i: number; terms: string[] }
const terms = (xs: (string | undefined)[]) => xs.filter((x): x is string => !!x).map(normalize)

export function createSearch(catalog: CatalogEntry[], diseases: Disease[], categories: Category[]) {
  const docs: Doc[] = [
    ...catalog.map((d, i) => ({ kind: 'drug' as const, i, terms: terms([d.name.fa, d.name.en, d.class?.fa, d.class?.en, ...d.brands, ...d.synonyms]) })),
    ...diseases.map((d, i) => ({ kind: 'disease' as const, i, terms: terms([d.name.fa, d.name.en, ...d.synonyms, ...d.icd10]) })),
    ...categories.map((c, i) => ({ kind: 'category' as const, i, terms: terms([c.name.fa, c.name.en]) })),
  ]
  const fuse = new Fuse(docs, { keys: ['terms'], threshold: 0.35, ignoreLocation: true })
  return (q: string, limit = 8): SearchResults => {
    const res: SearchResults = { drugs: [], diseases: [], categories: [] }
    const nq = normalize(q)
    if (!nq) return res
    for (const { item } of fuse.search(nq)) {
      if (item.kind === 'drug' && res.drugs.length < limit) res.drugs.push(catalog[item.i])
      if (item.kind === 'disease' && res.diseases.length < limit) res.diseases.push(diseases[item.i])
      if (item.kind === 'category' && res.categories.length < limit) res.categories.push(categories[item.i])
    }
    return res
  }
}
```
